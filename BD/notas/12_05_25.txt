Relação entre tabelas 

CREATE DATABASE joao_agenda;
USE joao_agenda;
CREATE TABLE contatos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    endereco VARCHAR(255),
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE telefones (
    id INT AUTO_INCREMENT PRIMARY KEY,
    contato_id INT NOT NULL,
    numero VARCHAR(20) NOT NULL,
    tipo ENUM('residencial', 'comercial', 'celular') NOT NULL,
    FOREIGN KEY (contato_id) REFERENCES contatos(id) ON DELETE CASCADE
);

SELECT * FROM contatos AS c , telefones AS t 
WHERE t.id = 1;

SELECT c.nome, COUNT(t.id) AS n_tel
FROM contatos AS c 
LEFT JOIN telefones AS t
ON c.id = t.contato_id
GROUP BY c.id;


SELECT * FROM contatos AS c 
LEFT JOIN telefones AS t 
ON c.id = t.contato_id;

SELECT *
FROM contatos 
WHERE nome LIKE "%Joao%" OR telefone LIKE "%37%" OR email LIKE "%@gmail%"
